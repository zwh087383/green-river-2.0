cmake_minimum_required(VERSION 3.16)
project(greenriver_native_motion)
if(NOT ROBOT_WS)
  set(ROBOT_WS "$ENV{HOME}/wheeled_robot_pkg")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/check_motion_sdk.py"
    --robot-ws "${ROBOT_WS}" --output "${CMAKE_CURRENT_BINARY_DIR}/sdk-contract.json"
  RESULT_VARIABLE SDK_CHECK_RESULT OUTPUT_VARIABLE SDK_CHECK_OUTPUT ERROR_VARIABLE SDK_CHECK_ERROR)
if(NOT SDK_CHECK_RESULT EQUAL 0)
  message(FATAL_ERROR "${SDK_CHECK_ERROR}")
endif()
message(STATUS "Motion SDK compile/link contract validated")
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(wheeled_robot_msgs REQUIRED)
find_path(GREENRIVER_JSON_INCLUDE_DIR nlohmann/json.hpp)
if(NOT GREENRIVER_JSON_INCLUDE_DIR)
  message(FATAL_ERROR "nlohmann/json.hpp is required for motion preset loading")
endif()
add_library(greenriver_native_motion SHARED native_motion_controller.cpp)
target_compile_features(greenriver_native_motion PUBLIC cxx_std_17)
target_compile_options(greenriver_native_motion PRIVATE -Wall -Wextra -Werror)
target_include_directories(greenriver_native_motion PRIVATE "${ROBOT_WS}/install/include" "${GREENRIVER_JSON_INCLUDE_DIR}")
ament_target_dependencies(greenriver_native_motion controller_interface pluginlib rclcpp wheeled_robot_msgs)
target_link_libraries(greenriver_native_motion "${ROBOT_WS}/install/lib/libarm_control.so" "${ROBOT_WS}/install/lib/libtrunk_control.so" "${ROBOT_WS}/install/lib/libeu_harmonic.so" pthread)
target_link_options(greenriver_native_motion PRIVATE "-Wl,-z,defs")
pluginlib_export_plugin_description_file(controller_interface native_motion_plugin.xml)
install(TARGETS greenriver_native_motion LIBRARY DESTINATION lib)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/sdk-contract.json" DESTINATION share/${PROJECT_NAME})
ament_package()
